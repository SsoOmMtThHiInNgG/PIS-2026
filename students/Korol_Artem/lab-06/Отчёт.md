            .withUsername("test")
            .withPassword("test");

        registry.add("spring.datasource.username", postgres::getUsername);
        registry.add("spring.datasource.password", postgres::getPassword);

    @Autowired

        Ticket ticket2 = new Ticket(null, 2L, 100L, 15);
        ticketRepository.save(ticket1);
        ticketRepository.save(ticket2);
        
        List<Ticket> tickets = ticketRepository.findByUserId(100L);
        
        assertEquals(2, tickets.size());
    }
}

### 3. E2E-тесты

**Сценарий:**
1. GET /api/events → получить мероприятия
2. POST /api/tickets → создать билет
3. GET /api/schedule/{eventId} → получить расписание
4. POST /api/events/{id}/publish → опубликовать мероприятие

**Код тестов:**

```java
// FullFlowE2ETest.java
@SpringBootTest(webEnvironment = SpringBootTest.WebEnvironment.RANDOM_PORT)
public class FullFlowE2ETest {

    @LocalServerPort
    private int port;
    
    @Autowired
    private TestRestTemplate restTemplate;
    
    @Test
    void fullUserFlowShouldWork() {
        String baseUrl = "http://localhost:" + port;
        
        // 1. Получить мероприятия
        ResponseEntity<Event[]> eventsResponse = restTemplate.getForEntity(
            baseUrl + "/api/events", Event[].class
        );
        assertTrue(eventsResponse.getStatusCode().is2xxSuccessful());
        
        // 2. Создать билет
        CreateTicketRequest request = new CreateTicketRequest(1L, 100L, 42);
        ResponseEntity<Ticket> ticketResponse = restTemplate.postForEntity(
            baseUrl + "/api/tickets", request, Ticket.class
        );
        assertEquals(HttpStatus.CREATED, ticketResponse.getStatusCode());
        assertNotNull(ticketResponse.getBody().getId());
        
        // 3. Получить расписание
        ResponseEntity<Schedule> scheduleResponse = restTemplate.getForEntity(
            baseUrl + "/api/schedule/1", Schedule.class
        );
        assertTrue(scheduleResponse.getStatusCode().is2xxSuccessful());
        
        // 4. Опубликовать мероприятие
        restTemplate.postForEntity(baseUrl + "/api/events/1/publish", null, Void.class);
        
        ResponseEntity<Event> eventResponse = restTemplate.getForEntity(
            baseUrl + "/api/events/1", Event.class
        );
        assertEquals(EventStatus.PUBLISHED, eventResponse.getBody().getStatus());
    }
}
```
